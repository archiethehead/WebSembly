;Count all powers of 2 up to R1

MOV R1, #10000

LOOP:

    ADD R0, R0, #1
    
    SUB R2, R0, #1
    AND R3, R0, R2

    CMP R3, #0
    BNE SKIP
    BEQ INCREMENT

    INCREMENT:
        ADD R4, R4, #1
	
    SKIP:
        CMP R0, R1
        BNE LOOP

STR R4, 0
HALT

;Set memory backwards from index

MOV R0, #1023

LOOP:
    
    STR R0, R1
    SUB R0, R0, #1

    ADD R1, R1, #1
    CMP R1, #1023
    BNE LOOP

HALT

;Count number of bits

MOV R1, #10

LOOP:
    
    CMP R1, #0
    BEQ FINAL-COUNT
    
    SUB R2, R1, #1
    AND R1, R2, R1
    
    ADD R0, R0, #1
    B LOOP

    FINAL-COUNT:
        STR R0, 0

HALT
